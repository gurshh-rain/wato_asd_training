cmake_minimum_required(VERSION 3.10)
project(moving_obstacles)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)

if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  foreach(_target rclcpp::rclcpp rcutils::rcutils)
    if(TARGET ${_target})
      get_target_property(_interface_libs ${_target} INTERFACE_LINK_LIBRARIES)
      if(_interface_libs)
        list(REMOVE_ITEM _interface_libs "atomic" "-latomic")
        set_target_properties(${_target} PROPERTIES INTERFACE_LINK_LIBRARIES "${_interface_libs}")
      endif()
    endif()
  endforeach()
endif()

add_executable(moving_obstacles_node src/moving_obstacles_node.cpp)
ament_target_dependencies(moving_obstacles_node rclcpp geometry_msgs rosgraph_msgs tf2_msgs)
install(TARGETS moving_obstacles_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
